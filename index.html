-- SSA Projects Tracker Database Schema -- Run this in Supabase SQL Editor -- Create team members table create table if not exists public.ssa_team_members ( id bigserial primary key, email text unique not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); alter table public.ssa_team_members enable row level security; create policy "Team members can view all members" on public.ssa_team_members for select using (auth.jwt() ->> 'email' in (select email from public.ssa_team_members)); -- Create projects table create table if not exists public.ssa_projects ( id bigserial primary key, name text not null, project_type text default 'Team', owner text, status text default 'Planned', target_hours integer, due_date date, notes text, year integer default 2026, created_by text not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null, updated_at timestamp with time zone default timezone('utc'::text, now()) not null ); alter table public.ssa_projects enable row level security; create policy "Approved team members can view projects" on public.ssa_projects for select using (auth.jwt() ->> 'email' in (select email from public.ssa_team_members)); create policy "Approved team members can create projects" on public.ssa_projects for insert with check (auth.jwt() ->> 'email' in (select email from public.ssa_team_members)); create policy "Team members can update projects" on public.ssa_projects for update using (auth.jwt() ->> 'email' in (select email from public.ssa_team_members)); -- Create time entries table create table if not exists public.ssa_time_entries ( id bigserial primary key, project_id bigint references public.ssa_projects(id) on delete cascade, hours decimal(5,2) not null, date_worked date not null, notes text, logged_by text not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null, updated_at timestamp with time zone default timezone('utc'::text, now()) not null ); alter table public.ssa_time_entries enable row level security; create policy "Approved team can view time entries" on public.ssa_time_entries for select using (auth.jwt() ->> 'email' in (select email from public.ssa_team_members)); create policy "Users can create own time entries" on public.ssa_time_entries for insert with check (auth.jwt() ->> 'email' in (select email from public.ssa_team_members) and logged_by = (auth.jwt() ->> 'email')); create policy "Users can update own time entries" on public.ssa_time_entries for update using (auth.jwt() ->> 'email' = logged_by); create policy "Users can delete own time entries" on public.ssa_time_entries for delete using (auth.jwt() ->> 'email' = logged_by); -- Create indexes for performance create index if not exists idx_projects_year on public.ssa_projects(year); create index if not exists idx_projects_status on public.ssa_projects(status); create index if not exists idx_time_entries_project on public.ssa_time_entries(project_id);
