async function logTime() { const loggerName = document.getElementById('loggerName').value.trim(); const date = document.getElementById('logDate').value; const hours = parseFloat(document.getElementById('logHours').value); const notes = document.getElementById('logNotes').value; if (!loggerName || !date || !hours) { alert('Fill in required fields (Name, Date, Hours)'); return; } if (currentEditingTimeEntryId) { // Edit mode const entry = timeEntries.find(e => e.id == currentEditingTimeEntryId); if (!entry) return; entry.logged_by = loggerName; entry.date_worked = date; entry.hours = hours; entry.notes = notes; if (testMode) { saveData(); } else if (db) { try { await db.from('ssa_time_entries').update(entry).eq('id', entry.id); } catch (err) { alert('Error: ' + err.message); } } currentEditingTimeEntryId = null; } else { // Add new entry (always create new, never merge) timeEntries.push({ id: Date.now(), project_id: currentProjectId, date_worked: date, hours, notes, logged_by: loggerName }); if (testMode) { saveData(); } else if (db) { try { await db.from('ssa_time_entries').insert([timeEntries[timeEntries.length - 1]]); } catch (err) { alert('Error: ' + err.message); } } } closeModal('timeEntryModal'); }
